registerForm.addEventListener('submit', function(e) { e.preventDefault(); const nombre = document.getElementById('nombre').value; const correo = document.getElementById('correo').value; const institucion = document.getElementById('institucion').value; // Enviamos los datos al servidor mediante AJAX/Fetch fetch('guardar_usuario.php', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: `nombre=${encodeURIComponent(nombre)}&correo=${encodeURIComponent(correo)}&institucion= ${encodeURIComponent(institucion)}` }) .then(response => response.json()) .then(data => { if(data.status === 'success') { // Guardamos localmente solo para mantener la sesión visualmente abierta localStorage.setItem('cla_nombre', nombre); localStorage.setItem('cla_institucion', institucion); // Mostramos el dashboard que ya tienes diseñado mostrarDashboard(nombre, institucion); } else { alert(data.message); } }) .catch(error => { console.error('Error:', error); // Si estás probando localmente en tu PC sin servidor PHP, permitimos el paso visual de todas formas: mostrarDashboard(nombre, institucion); }); });
